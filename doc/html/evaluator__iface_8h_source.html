58  {
59  size_t state_len = states.size() / n_blocks;
60  std::vector<double> state(state_len);
61  values.resize(n_blocks);
62  for (int i = 0; i < n_blocks; i++)
63  {
64  std::copy(states.begin() + i * state_len, states.begin() + (i + 1) * state_len, state.begin());
